\begin{frame}{From State Values to Action Values}
\small
The \term{state value function} \(\modelvar{V^\sigma(s)}\): follow the deterministic policy \(\modelvar{\sigma}\) now and afterward.
\[
V^\sigma(s)=
\underbrace{r\bigl(s,\sigma(s)\bigr)}_{\text{Immediate reward}}
+\underbrace{\beta\,\E\!\left[V^\sigma(s')\mid s,\ a=\sigma(s)\right]}_{\text{Discounted continuation value}}.
\]
\par\medskip
The \term{state-action value function} \(\modelvar{Q^\sigma(s,a)}\): choose \(\modelvar{a}\) now, then follow \(\modelvar{\sigma}\).
\[
Q^\sigma(s,a)=
\underbrace{r(s,a)}_{\text{Immediate reward}}
+\underbrace{\beta\,\E\!\left[V^\sigma(s')\mid s,a\right]}_{\text{Discounted continuation value}}.
\]
\par\medskip
The first action \(\modelvar{a}\) is deliberately chosen; it need not be an action that \(\modelvar{\sigma}\) selects at \(\modelvar{s}\).
\end{frame}

\begin{frame}{Linking Q and V}
Suppose the first action \(\modelvar{a}\) is also chosen according to \(\modelvar{\sigma}\). Then the policy governs the entire rollout.
\begin{itemize}
\item \textbf{Deterministic policy:} \(\modelvar{a=\sigma(s)}\), so
\[V^\sigma(s)=Q^\sigma(s,\sigma(s)).\]
\item \textbf{Randomized policy:} \(\modelvar{a\sim\sigma(s)}\). For a finite action set,
\[V^\sigma(s)=\sum_{a\in\Aset(s)}\Pr\{\sigma(s)=a\}\,Q^\sigma(s,a).\]
\end{itemize}
\end{frame}

\begin{frame}{Greedy Policy Improvement}
\small
For the finite-state results, assume finite feasible action sets, bounded rewards, \(\modelvar{0<\beta<1}\), and time-homogeneous dynamics. Policies are stationary and deterministic.
\par\medskip
Given the exact value \(\modelvar{V^\sigma}\), choose a \term{greedy policy} at every state:
\[\widetilde\sigma(s)\in\argmax_{a\in\Aset(s)}Q^\sigma(s,a).\]
Because the old action remains feasible,
\[Q^\sigma(s,\widetilde\sigma(s))\geq Q^\sigma(s,\sigma(s))=V^\sigma(s),
\qquad \forall s.\]
Greedy choices maximize immediate reward \textbf{plus continuation value}, rather than immediate reward alone.
\par\medskip
\term{Policy improvement theorem:} if the displayed inequality holds at every state, then
\[V^{\widetilde\sigma}(s)\geq V^\sigma(s),\qquad\forall s.\]
\end{frame}

\begin{frame}{Why Policy Improvement Works}
\small
For any candidate continuation value \(\modelvar{v}\), the \term{policy evaluation operator} for \(\modelvar{\widetilde\sigma}\) is
\[(T_{\widetilde\sigma}v)(s)=r(s,\widetilde\sigma(s))
+\beta\E[v(s')\mid s,a=\widetilde\sigma(s)].\]
It is \term{monotone}: \(\modelvar{v\leq w}\) implies \(\modelvar{T_{\widetilde\sigma}v\leq T_{\widetilde\sigma}w}\), state by state.
\par\medskip
The improvement condition is \(\modelvar{V^\sigma\leq T_{\widetilde\sigma}V^\sigma}\). Applying the same operator repeatedly gives
\[V^\sigma\leq T_{\widetilde\sigma}V^\sigma
\leq T_{\widetilde\sigma}^{\,2}V^\sigma\leq\cdots
\longrightarrow V^{\widetilde\sigma}.\]
\begin{itemize}
\item Each update follows the new policy for one more step; the influence of the initial continuation vanishes at rate \(\modelvar{\beta^n}\).
\item A strict improvement in the first inequality at a state gives a strict value gain at that state.
\end{itemize}
\end{frame}

\begin{frame}{The Principle of Optimality}
\small
Define the \term{optimal values} over feasible policies:
\[V^*(s)=\max_\sigma V^\sigma(s),\qquad
Q^*(s,a)=\max_\sigma Q^\sigma(s,a).\]
In this setting, a stationary deterministic policy is optimal among all admissible policies.
\par\medskip
\term{Principle of optimality:} an optimal policy must have an optimal continuation from every state reached with positive probability.
\begin{align*}
Q^*(s,a)&=r(s,a)+\beta\E[V^*(s')\mid s,a],\\
V^*(s)&=\max_{a\in\Aset(s)}Q^*(s,a).
\end{align*}
Combining them gives the \term{Bellman optimality equation}:
\[V^*(s)=\max_{a\in\Aset(s)}\{r(s,a)+\beta\E[V^*(s')\mid s,a]\}.\]
\end{frame}

\begin{frame}{The Bellman Optimality Operator}
Replace the optimal continuation by a candidate \(\modelvar{v}\). The \term{Bellman optimality operator} is
\[(Tv)(s)=\max_{a\in\Aset(s)}\left\{r(s,a)
+\beta\sum_{s'\in\Sset}P(s'\mid s,a)v(s')\right\}.\]
\begin{itemize}
\item \(\modelvar{T_\sigma v}\) uses the prescribed action \(\modelvar{\sigma(s)}\); \(\modelvar{Tv}\) chooses the best action for continuation \(\modelvar{v}\).
\item A policy greedy with respect to \(\modelvar{v}\) attains this maximum: \(\modelvar{T_\sigma v=Tv}\).
\item The optimal value satisfies \(\modelvar{TV^*=V^*}\): it is a \term{fixed point} of \(\modelvar{T}\).
\end{itemize}
\end{frame}

\begin{frame}{Contraction and Uniqueness}
\small
Use \(\modelvar{\|v-w\|_\infty=\max_s|v(s)-w(s)|}\). For the same state and action,
\[\left|\beta\E[v(s')-w(s')\mid s,a]\right|\leq\beta\|v-w\|_\infty.\]
Taking maxima cannot enlarge this bound, since
\[|\max_a x_a-\max_a y_a|\leq\max_a|x_a-y_a|.\]
Therefore \(\modelvar{T}\) is a \term{contraction}:
\[\|Tv-Tw\|_\infty\leq\beta\|v-w\|_\infty.\]
The contraction mapping theorem gives a unique fixed point, \(\modelvar{V^*}\), and convergence to it from any initial value vector. The same bound holds for each fixed-policy operator \(\modelvar{T_\sigma}\).
\end{frame}

\begin{frame}{When Is a Policy Optimal?}
\small
The \term{one-step deviation principle} gives a test using the policy's own value:
\[\sigma(s)\in\argmax_{a\in\Aset(s)}Q^\sigma(s,a)\quad\forall s
\quad\Longleftrightarrow\quad V^\sigma=V^*.\]
\begin{itemize}
\item \textbf{Sufficiency:} if no first-action change improves the return at any state, then
\[V^\sigma=T_\sigma V^\sigma=TV^\sigma.\]
The Bellman fixed point is unique, so \(\modelvar{V^\sigma=V^*}\).
\item \textbf{Necessity:} if \(\modelvar{V^\sigma=V^*}\), then
\[T_\sigma V^*=V^*=TV^*,\]
so \(\modelvar{\sigma}\) attains the maximum at every state.
\end{itemize}
The comparison checks \textbf{all feasible actions at every state}, with the exact continuation value.
\end{frame}

\begin{frame}{Policy Iteration}
\small
Start from any feasible policy \(\modelvar{\sigma_0}\). Repeat:
\begin{enumerate}
\item \term{Policy evaluation:} solve exactly for the current policy's value,
\[V^{\sigma_k}=T_{\sigma_k}V^{\sigma_k}.\]
\item \term{Policy improvement:} choose
\[\sigma_{k+1}(s)\in\argmax_{a\in\Aset(s)}Q^{\sigma_k}(s,a),\qquad\forall s.\]
Keep the current action whenever it is a maximizer.
\item If \(\modelvar{\sigma_{k+1}=\sigma_k}\), stop: the policy is optimal. Otherwise evaluate the new policy and repeat.
\end{enumerate}
\[V^{\sigma_0}\leq V^{\sigma_1}\leq V^{\sigma_2}\leq\cdots\leq V^*.\]
Every nonoptimal policy is strictly improved at some state. There are finitely many deterministic policies, so the algorithm reaches an optimal policy in finitely many iterations.
\end{frame}

\begin{frame}{Value Function Iteration}
\small
VFI updates a value estimate \(\modelvar{v_k}\), without evaluating a policy to completion.
\begin{enumerate}
\item Initialize \(\modelvar{v_0}\), for example at zero.
\item Using the old values, compute the scores and the new values:
\begin{align*}
Q_k(s,a)&=r(s,a)+\beta\E[v_k(s')\mid s,a],\\
v_{k+1}(s)&=\max_{a\in\Aset(s)}Q_k(s,a)=(Tv_k)(s).
\end{align*}
\item Repeat these updates until the required accuracy is reached.
\end{enumerate}
A \term{sweep} updates all states using the same old \(\modelvar{v_k}\); store the results in a separate array.
\par\medskip
Contraction gives
\[\|v_k-V^*\|_\infty\leq\beta^k\|v_0-V^*\|_\infty\longrightarrow0.\]
\(\modelvar{v_k}\) need not equal any \(\modelvar{V^\sigma}\), and \(\modelvar{Q_k}\) need not equal any \(\modelvar{Q^\sigma}\). VFI convergence follows from contraction, not successive exact policy evaluations.
\end{frame}

\begin{frame}{Stopping VFI and Recovering a Policy}
\small
After computing \(\modelvar{v_{k+1}=Tv_k}\), measure the \term{Bellman residual}:
\[d_k=\|Tv_k-v_k\|_\infty=\|v_{k+1}-v_k\|_\infty.\]
Contraction implies
\[\|v_k-V^*\|_\infty\leq d_k+\beta\|v_k-V^*\|_\infty,
\quad\text{so}\quad\|v_k-V^*\|_\infty\leq\frac{d_k}{1-\beta}.\]
\begin{itemize}
\item For value error at most \(\modelvar{\varepsilon}\), stop when \(\modelvar{d_k\leq(1-\beta)\varepsilon}\) and return \(\modelvar{v_k}\).
\item Recover a policy from the scores computed using this returned value:
\[\widehat\sigma(s)\in\argmax_{a\in\Aset(s)}Q_k(s,a).\]
\item Evaluate \(\modelvar{V^{\widehat\sigma}}\) to check the policy's achieved return. Intermediate greedy policies in VFI need not improve monotonically.
\end{itemize}
\footnotesize The bound assumes exact expectations and maximization; numerical approximations add errors.
\end{frame}

\begin{frame}{Example: VFI for McCall Job Search}
\small
The state is \(\modelvar{(\mathrm U,w)}\) or \(\modelvar{(\mathrm E,w)}\). Rejecting pays \(\modelvar{b}\) and draws a new offer \(\modelvar{w'}\) with probability \(\modelvar{q(w')}\); accepting pays \(\modelvar{w}\) forever.
\[v_k((\mathrm E,w))=\frac{w}{1-\beta}\quad\text{for every }k.\]
Use this known employed value and update unemployed states:
\begin{align*}
Q_k((\mathrm U,w),\text{accept})&=\frac{w}{1-\beta},\\
Q_k((\mathrm U,w),\text{reject})&=b+\beta\sum_{w'}q(w')v_k((\mathrm U,w')),\\
v_{k+1}((\mathrm U,w))&=\max\{Q_k((\mathrm U,w),\text{accept}),\ Q_k((\mathrm U,w),\text{reject})\}.
\end{align*}
Initialize \(\modelvar{v_0((\mathrm U,w))=0}\). Employed-state values stay fixed at their known values; only unemployed-state values need iteration.
\end{frame}

\begin{frame}{Example: VFI for Continuous Saving}
\small
With storage capacity \(\modelvar{M\geq0}\), stock is \(\modelvar{s\in[0,M+B]}\). Choose saving \(\modelvar{a\in[0,\min(s,M)]}\) and consume \(\modelvar{c=s-a}\).
\begin{itemize}
\item Income \(\modelvar{Y'\sim\mathrm{Uniform}[0,B]}\), \(\modelvar{B>0}\), is independent across periods; next stock is \(\modelvar{s'=a+Y'}\).
\item Reward is \(\modelvar{r(s,a)=(s-a)^\alpha}\), with \(\modelvar{0<\alpha<1}\).
\end{itemize}
The update averages over income and maximizes over saving:
\begin{align*}
Q_k(s,a)&=(s-a)^\alpha+\frac{\beta}{B}\int_0^B v_k(a+y)\,dy,\\
v_{k+1}(s)&=\max_{a\in[0,\min(s,M)]}Q_k(s,a).
\end{align*}
\small Assume \(\modelvar{0<\beta<1}\). The state and feasible action sets are compact, and rewards are bounded. The exact operator maps continuous functions to continuous functions and is a contraction in the supremum norm.
\end{frame}

\begin{frame}{Computing the Continuous Update}
To implement the continuous saving model numerically:
\begin{enumerate}
\item \textbf{Represent \(\modelvar{v_k}\):} store values at stock points and \term{interpolate} between them.
\item \textbf{Compute the expectation:} approximate the income integral using \term{numerical quadrature} or sampled incomes.
\item \textbf{Choose saving:} maximize over the feasible interval using numerical optimization or an action grid.
\end{enumerate}
\par\medskip
These approximations add errors beyond the VFI stopping error. Refine the stock grid, integration, and maximization to check that values and decisions are stable.
\par\smallskip
The model remains continuous. Finite termination of policy iteration for finite MDPs does not automatically apply to this model.
\end{frame}
