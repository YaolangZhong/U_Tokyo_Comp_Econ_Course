\begin{frame}{From State Values to Action Values}
\small
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
The \term{state value function} \(\modelvar{V^\sigma(s)}\): follow the deterministic policy \(\modelvar{\sigma}\) now and afterward.
\[
V^\sigma(s)=
\underbrace{r\bigl(s,\sigma(s)\bigr)}_{\text{Immediate reward}}
+\underbrace{\beta\,\E\!\left[V^\sigma(s')\mid s,\ a=\sigma(s)\right]}_{\text{Discounted continuation value}}.
\]
\par\smallskip
The \term{state-action value function} \(\modelvar{Q^\sigma(s,a)}\): choose \(\modelvar{a}\) now, then follow \(\modelvar{\sigma}\).
\[
Q^\sigma(s,a)=
\underbrace{r(s,a)}_{\text{Immediate reward}}
+\underbrace{\beta\,\E\!\left[V^\sigma(s')\mid s,a\right]}_{\text{Discounted continuation value}}.
\]
The first action \(\modelvar{a}\) need not be the action selected by \(\modelvar{\sigma}\).
\par\smallskip
If the first action also follows \(\modelvar{\sigma}\), then
\begin{align*}
\text{Deterministic:}\quad V^\sigma(s)&=Q^\sigma(s,\sigma(s)),\\[2pt]
\text{Randomized:}\quad V^\sigma(s)&=\sum_{a\in\Aset(s)}\Pr\{\sigma(s)=a\}\,Q^\sigma(s,a).
\end{align*}
\end{frame}

\begin{frame}{Greedy Policy Improvement}
\small
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
For a given policy \(\modelvar{\sigma}\), let \(\modelvar{V^\sigma}\) and \(\modelvar{Q^\sigma}\) denote its state and state--action value functions. At a particular state \(\modelvar{s}\), a greedy \term{policy improvement step} defines the new policy \(\modelvar{\widetilde\sigma}\) by
\[
\widetilde a\in\argmax_{a\in\Aset(s)}Q^\sigma(s,a),
\qquad
\widetilde\sigma(x)=
\begin{cases}
\widetilde a,&x=s,\\
\sigma(x),&x\neq s.
\end{cases}
\]
i.e., the new policy uses \(\modelvar{\widetilde a}\) at \(\modelvar{s}\) and keeps \(\modelvar{\sigma}\) at every other state.
\par\medskip
\term{Policy improvement theorem.} More generally, let \(\modelvar{\sigma}\) and \(\modelvar{\widetilde\sigma}\) be any two stationary deterministic policies. Assume \(\modelvar{|r(s,a)|\leq\bar r<\infty}\) for all feasible \(\modelvar{(s,a)}\), and \(\modelvar{0<\beta<1}\). If
\[Q^\sigma(s,\widetilde\sigma(s))\geq V^\sigma(s)\qquad\forall s\in\Sset,\]
then
\[V^{\widetilde\sigma}(s)\geq V^\sigma(s)\qquad\forall s\in\Sset.\]
If, at every state \(\modelvar{s}\), taking \(\modelvar{\widetilde\sigma(s)}\) once and then following \(\modelvar{\sigma}\) is no worse than following \(\modelvar{\sigma}\) throughout, then following \(\modelvar{\widetilde\sigma}\) at every step is also no worse.
\end{frame}

\begin{frame}{The Principle of Optimality}
\small
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
Let \(\modelvar{\sigma^*}\) be an \term{optimal policy}, with associated values
\[V^*(s)=V^{\sigma^*}(s)=\max_\sigma V^\sigma(s),\qquad
Q^*(s,a)=Q^{\sigma^*}(s,a).\]
\term{Principle of optimality:} an optimal policy must also be optimal for every remaining \textbf{subproblem}. After reaching \(\modelvar{s'}\), the same \(\modelvar{\sigma^*}\) solves the remaining problem starting at \(\modelvar{s'}\), with value \(\modelvar{V^*(s')}\).
\par\smallskip
\textbf{1. Fix the first action \(\modelvar{a}\).} Receive \(\modelvar{r(s,a)}\), then solve the next-state subproblem optimally. Averaging over possible next states gives
\[Q^*(s,a)=r(s,a)+\beta\E[V^*(s')\mid s,a].\]
\textbf{2. Choose the best first action.} The optimal policy \(\modelvar{\sigma^*}\) attains the maximum:
\[V^*(s)=Q^*(s,\sigma^*(s))=\max_{a\in\Aset(s)}Q^*(s,a).\]
Substituting the first equation into the second gives the \term{Bellman optimality equation}:
\[V^*(s)=\max_{a\in\Aset(s)}\{r(s,a)+\beta\E[V^*(s')\mid s,a]\}.\]
\end{frame}

\begin{frame}{The Bellman Optimality Operator}
\small
The Bellman equation characterizes \(\modelvar{V^*}\), but \(\modelvar{V^*}\) is still unknown. To compute it, start with a candidate continuation-value function \(\modelvar{V}\).
\par\medskip
The \term{Bellman optimality operator} \(\modelvar{T}\) maps \(\modelvar{V}\) to an updated value function \(\modelvar{TV}\):
\[(TV)(s)=\max_{a\in\Aset(s)}\{r(s,a)+\beta\E[V(s')\mid s,a]\}.\]
At each state, evaluate each first action using \(\modelvar{V}\) for the continuation, then keep the highest value.
\par\medskip
At the optimal value, the update changes nothing:
\[TV^*=V^*.\]
Thus \(\modelvar{V^*}\) is a \term{fixed point} of \(\modelvar{T}\).
\par\medskip
This suggests repeated updates, \(\modelvar{V_{k+1}=TV_k}\). We need to establish that they converge to \(\modelvar{V^*}\), regardless of the initial guess \(\modelvar{V_0}\).
\end{frame}

\begin{frame}{Convergence of Bellman Updates}
\small
To justify repeated updates, compare how \(\modelvar{T}\) transforms two candidate value functions \(\modelvar{V_1,V_2}\). Their largest difference across states is
\[\|V_1-V_2\|_\infty=\max_s|V_1(s)-V_2(s)|.\]
Averaging cannot increase this difference; discounting multiplies it by \(\modelvar{\beta}\). Maximizing over actions preserves the bound:
\[\|TV_1-TV_2\|_\infty\leq\beta\|V_1-V_2\|_\infty,\qquad 0<\beta<1.\]
This is the \term{contraction property}: the largest difference after an update is at most \(\modelvar{\beta}\) times its previous size.
\par\medskip
The contraction mapping theorem therefore gives:
\begin{itemize}
\item \textbf{A unique solution.} \(\modelvar{V^*}\) is the only fixed point of \(\modelvar{T}\), hence the only solution to the Bellman optimality equation.
\item \textbf{Convergence.} Starting from any \(\modelvar{V_0}\), the updates \(\modelvar{V_{k+1}=TV_k}\) satisfy
\[\|V_k-V^*\|_\infty\leq\beta^k\|V_0-V^*\|_\infty\longrightarrow0.\]
\end{itemize}
\end{frame}

\begin{frame}{When Is a Policy Optimal?}
\small
The \term{one-step deviation principle} gives a test using the policy's own value:
\[\sigma(s)\in\argmax_{a\in\Aset(s)}Q^\sigma(s,a)\quad\forall s
\quad\Longleftrightarrow\quad V^\sigma=V^*.\]
\begin{itemize}
\item \textbf{Sufficiency:} if no first-action change improves the return at any state, then
\[V^\sigma(s)=Q^\sigma(s,\sigma(s))=\max_a Q^\sigma(s,a)=(TV^\sigma)(s).\]
The Bellman fixed point is unique, so \(\modelvar{V^\sigma=V^*}\).
\item \textbf{Necessity:} if \(\modelvar{V^\sigma=V^*}\), then
\[Q^\sigma=Q^*,\qquad Q^\sigma(s,\sigma(s))=V^*(s)=\max_a Q^\sigma(s,a),\]
so \(\modelvar{\sigma}\) attains the maximum at every state.
\end{itemize}
\end{frame}

\begin{frame}{Policy Iteration}
\small
Start from any feasible policy \(\modelvar{\sigma_0}\). Repeat:
\begin{enumerate}
\item \term{Policy evaluation:} solve exactly for the current policy's value,
\[V^{\sigma_k}(s)=r(s,\sigma_k(s))+\beta\E[V^{\sigma_k}(s')\mid s,a=\sigma_k(s)].\]
\item \term{Policy improvement:} choose
\[\sigma_{k+1}(s)\in\argmax_{a\in\Aset(s)}Q^{\sigma_k}(s,a),\qquad\forall s.\]
Keep the current action whenever it is a maximizer.
\item If \(\modelvar{\sigma_{k+1}=\sigma_k}\), stop: the policy is optimal. Otherwise evaluate the new policy and repeat.
\end{enumerate}
\[V^{\sigma_0}\leq V^{\sigma_1}\leq V^{\sigma_2}\leq\cdots\leq V^*.\]
Each nonoptimal policy improves strictly at some state. With finitely many policies, this process terminates at an optimum.
\end{frame}

\begin{frame}{Value Function Iteration}
\small
VFI updates a value estimate \(\modelvar{V_k}\), without evaluating a policy to completion.
\begin{enumerate}
\item Initialize \(\modelvar{V_0}\), for example at zero.
\item Using the old values, compute the scores and the new values:
\begin{align*}
Q_k(s,a)&=r(s,a)+\beta\E[V_k(s')\mid s,a],\\
V_{k+1}(s)&=\max_{a\in\Aset(s)}Q_k(s,a)=(TV_k)(s).
\end{align*}
\item Repeat these updates until the required accuracy is reached.
\end{enumerate}
A \term{sweep} updates all states using the same old \(\modelvar{V_k}\); store the results in a separate array.
\par\medskip
The estimates \(\modelvar{V_k}\) and \(\modelvar{Q_k}\) need not be the values of a particular policy.
\end{frame}

\begin{frame}{Stopping VFI and Recovering a Policy}
\small
After computing \(\modelvar{V_{k+1}=TV_k}\), measure the \term{Bellman residual}:
\[d_k=\|TV_k-V_k\|_\infty=\|V_{k+1}-V_k\|_\infty.\]
Contraction implies
\[\|V_k-V^*\|_\infty\leq d_k+\beta\|V_k-V^*\|_\infty,
\quad\text{so}\quad\|V_k-V^*\|_\infty\leq\frac{d_k}{1-\beta}.\]
\begin{itemize}
\item For value error at most \(\modelvar{\varepsilon}\), stop when \(\modelvar{d_k\leq(1-\beta)\varepsilon}\) and return \(\modelvar{V_k}\).
\item Recover a policy from the scores computed using this returned value:
\[\widehat\sigma(s)\in\argmax_{a\in\Aset(s)}Q_k(s,a).\]
\item Evaluate \(\modelvar{V^{\widehat\sigma}}\) to check the policy's achieved return. Intermediate greedy policies in VFI need not improve monotonically.
\end{itemize}
\end{frame}
