\begin{frame}{Lecture Overview}
\begin{itemize}
\item \textbf{From State Values to Action Values}
  \begin{itemize}\item Evaluate a maze policy and its state values\item Introduce \(\modelvar{Q^\sigma}\) to compare first actions and link it to \(\modelvar{V^\sigma}\)\end{itemize}
\item \textbf{The Principle of Optimality}
  \begin{itemize}\item Improve a continuation from a reached state\item Derive optimal state and action values\end{itemize}
\item \textbf{Value Function Iteration}
  \begin{itemize}\item Update values, check convergence, and recover a policy\item Apply the algorithm to job search and continuous saving\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Recap: Computing the Value Function of a Given Policy}
\small
Consider a specific deterministic policy \(\modelvar{\sigma}\).
\begin{columns}[c]
\begin{column}{.60\textwidth}
\begin{figure}\centering

\end{tikzpicture}\MazeLegend
\caption{Actions and path under \(\sigma\).}
\end{figure}
\end{column}
\begin{column}{.37\textwidth}
\begin{itemize}
\item U: up; D: down;\\L: left; R: right.
\item Each move gives \(\modelvar{-1}\), including entry into cell 8.
\item Set \(\modelvar{\beta=1}\) for simplicity.

\begin{frame}{From State Values to Action Values}
\small
The \term{state value function} \(\modelvar{V^\sigma(s)}\): follow the deterministic policy \(\modelvar{\sigma}\) now and afterward.
\[
V^\sigma(s)=
\underbrace{r\bigl(s,\sigma(s)\bigr)}_{\text{Immediate reward}}
+\underbrace{\beta\,\E\!\left[V^\sigma(s')\mid s,\ a=\sigma(s)\right]}_{\text{Discounted continuation value}}.
\]
\par\medskip
The \term{state-action value function} \(\modelvar{Q^\sigma(s,a)}\): choose \(\modelvar{a}\) now, then follow \(\modelvar{\sigma}\).
\[
Q^\sigma(s,a)=
\underbrace{r(s,a)}_{\text{Immediate reward}}
+\underbrace{\beta\,\E\!\left[V^\sigma(s')\mid s,a\right]}_{\text{Discounted continuation value}}.
\]
\par\medskip
The first action \(\modelvar{a}\) is deliberately chosen; it need not be an action that \(\modelvar{\sigma}\) selects at \(\modelvar{s}\).
\end{frame}

\begin{frame}{Linking Q and V}
Suppose the first action \(\modelvar{a}\) is also chosen according to \(\modelvar{\sigma}\). Then the policy governs the entire rollout.
\begin{itemize}
\item \textbf{Deterministic policy:} \(\modelvar{a=\sigma(s)}\), so
\[V^\sigma(s)=Q^\sigma(s,\sigma(s)).\]
\item \textbf{Randomized policy:} \(\modelvar{a\sim\sigma(s)}\). For a finite action set,
\[V^\sigma(s)=\sum_{a\in\Aset(s)}\Pr\{\sigma(s)=a\}\,Q^\sigma(s,a).\]
\end{itemize}
\end{frame}

\begin{frame}{Optimal State and Action Values}
For a finite discounted MDP, an optimal stationary deterministic policy exists. Define
\[V^*(s)=\max_\sigma V^\sigma(s),\qquad
Q^*(s,a)=\max_\sigma Q^\sigma(s,a).\]
The principle of optimality gives
\begin{align*}
Q^*(s,a)&=r(s,a)+\beta\,\E\!\left[V^*(s')\mid s,a\right],\\
V^*(s)&=\max_{a\in\Aset(s)}Q^*(s,a).
\end{align*}
\begin{itemize}
\item \(\modelvar{Q^*}\): fix the current action, then choose optimally afterward.
\item \(\modelvar{V^*}\): also choose the current action optimally.
\item The expectation averages the outcomes of the chosen action; it does not let us choose the random outcome.
\end{itemize}
\end{frame}

\begin{frame}{The Bellman Optimality Operator}
Replace the unknown optimal continuation value by a candidate function \(\modelvar{v}\). The \term{Bellman optimality operator} produces
\[(Tv)(s)=\max_{a\in\Aset(s)}\left\{
r(s,a)+\beta\sum_{s'}P(s'\mid s,a)v(s')\right\}.\]
\begin{itemize}
\item For each action, add the immediate reward to the expected discounted continuation value; take the largest result.
\item The optimal value is a \term{fixed point}: \(\modelvar{TV^*=V^*}\).
\end{itemize}
\small The sum describes discrete next states; use a probability integral for continuous next states. Unlike fixed-policy evaluation, \(\modelvar{T}\) maximizes over the current action.
\end{frame}

\begin{frame}{Value Function Iteration}
\small
Given a finite MDP, its reward function and transition kernel, and value-error tolerance \(\modelvar{\varepsilon>0}\):
\begin{enumerate}
\item Initialize \(\modelvar{v_0(s)=0}\) and set \(\modelvar{k=0}\). Keep zero-reward absorbing goal states at zero.
\item Score every feasible action using the \textbf{old} value function:
\[Q_k(s,a)=r(s,a)+\beta\,\E\!\left[v_k(s')\mid s,a\right].\]
\item Store the best score in a new array: \(\modelvar{v_{k+1}(s)=\max_a Q_k(s,a)}\).
\item Compute \(\modelvar{d_k=\max_s|v_{k+1}(s)-v_k(s)|}\). If \(\modelvar{d_k\leq(1-\beta)\varepsilon}\), return \(\modelvar{v_k}\); otherwise increase \(\modelvar{k}\) and repeat.
\end{enumerate}
\(\modelvar{Q_k}\) uses the approximate continuation \(\modelvar{v_k}\); it need not equal \(\modelvar{Q^\sigma}\) for a particular policy.
\par\smallskip
A \term{sweep} updates every state once. The separate new array ensures all updates use the same \(\modelvar{v_k}\).
\end{frame}

\begin{frame}{Why VFI Converges}
\small
For finite states, use \(\modelvar{\|v-w\|_\infty=\max_s|v(s)-w(s)|}\). The difference between two scores for the \textbf{same action} satisfies
\[\left|\beta\E[v(s')-w(s')\mid s,a]\right|\leq\beta\|v-w\|_\infty.\]
Since \(\modelvar{|\max_a x_a-\max_a y_a|\leq\max_a|x_a-y_a|}\),
\[\|Tv-Tw\|_\infty\leq\beta\|v-w\|_\infty.\]
Thus \(\modelvar{T}\) is a \term{contraction}: the new maximum distance is at most \(\modelvar{\beta}\) times the old distance.
\begin{itemize}
\item There is a unique fixed point \(\modelvar{V^*}\); every initial finite vector converges to it.
\item After \(\modelvar{k}\) sweeps,
\[\|v_k-V^*\|_\infty\leq\beta^k\|v_0-V^*\|_\infty.\]
\item A discount factor closer to one gives a slower worst-case rate.
\end{itemize}
\end{frame}

\begin{frame}{A Stopping Rule With an Error Bound}
\small
After computing \(\modelvar{v_{k+1}=Tv_k}\), measure the \term{Bellman residual} of \(\modelvar{v_k}\):
\[d_k=\|Tv_k-v_k\|_\infty=\|v_{k+1}-v_k\|_\infty.\]
The contraction inequality gives
\begin{align*}
\|v_k-V^*\|_\infty
&\leq\|v_k-Tv_k\|_\infty+\|Tv_k-TV^*\|_\infty\\
&\leq d_k+\beta\|v_k-V^*\|_\infty,
\end{align*}
so \(\modelvar{\|v_k-V^*\|_\infty\leq d_k/(1-\beta)}\).
\begin{itemize}
\item To guarantee value error at most \(\modelvar{\varepsilon}\), stop when \(\modelvar{d_k\leq(1-\beta)\varepsilon}\) and return \(\modelvar{v_k}\).
\item At \(\modelvar{\beta=0.9}\) and \(\modelvar{\varepsilon=0.01}\), require \(\modelvar{d_k\leq0.001}\).
\end{itemize}
\footnotesize This guarantee uses exact expectations and maximization. Numerical approximation introduces additional errors.
\end{frame}

\begin{frame}{Recovering a Policy}
Use the action scores calculated from the returned \(\modelvar{v_k}\). A \term{greedy policy} chooses
\[\widehat\sigma(s)\in\argmax_{a\in\Aset(s)}Q_k(s,a).\]
\begin{itemize}
\item If \(\modelvar{v_k=V^*}\), then \(\modelvar{Q_k=Q^*}\), and every maximizing action is optimal.
\item With approximate values, evaluate the resulting policy to check its achieved return.
\item If several actions tie, choose one consistently.
\end{itemize}
\par\medskip
VFI updates value estimates. The final maximization converts those estimates into a decision rule for each state.
\end{frame}

\begin{frame}{Example: VFI for McCall Job Search}
\small
The state is \(\modelvar{(\mathrm U,w)}\) or \(\modelvar{(\mathrm E,w)}\). Rejecting pays \(\modelvar{b}\) and draws a new offer \(\modelvar{w'}\) with probability \(\modelvar{q(w')}\); accepting pays \(\modelvar{w}\) forever.
\[v_k((\mathrm E,w))=\frac{w}{1-\beta}\quad\text{for every }k.\]
Use this known employed value and update unemployed states:
\begin{align*}
Q_k((\mathrm U,w),\text{accept})&=\frac{w}{1-\beta},\\
Q_k((\mathrm U,w),\text{reject})&=b+\beta\sum_{w'}q(w')v_k((\mathrm U,w')),\\
v_{k+1}((\mathrm U,w))&=\max\{Q_k((\mathrm U,w),\text{accept}),\ Q_k((\mathrm U,w),\text{reject})\}.
\end{align*}
Initialize \(\modelvar{v_0((\mathrm U,w))=0}\). Employed-state values stay fixed at their known values; only unemployed-state values need iteration.
\end{frame}

\begin{frame}{Example: VFI for Continuous Saving}
\small
With storage capacity \(\modelvar{M\geq0}\), stock is \(\modelvar{s\in[0,M+B]}\). Choose saving \(\modelvar{a\in[0,\min(s,M)]}\) and consume \(\modelvar{c=s-a}\).
\begin{itemize}
\item Income \(\modelvar{Y'\sim\mathrm{Uniform}[0,B]}\), \(\modelvar{B>0}\), is independent across periods; next stock is \(\modelvar{s'=a+Y'}\).
\item Reward is \(\modelvar{r(s,a)=(s-a)^\alpha}\), with \(\modelvar{0<\alpha<1}\).
\end{itemize}
The update averages over income and maximizes over saving:
\begin{align*}
Q_k(s,a)&=(s-a)^\alpha+\frac{\beta}{B}\int_0^B v_k(a+y)\,dy,\\
v_{k+1}(s)&=\max_{a\in[0,\min(s,M)]}Q_k(s,a).
\end{align*}
\small The exact operator is a contraction under the supremum norm on continuous value functions. The state interval is compact, rewards are bounded, and \(\modelvar{0<\beta<1}\).
\end{frame}

\begin{frame}{Computing the Continuous Update}
To implement the continuous saving model numerically:
\begin{enumerate}
\item \textbf{Represent \(\modelvar{v_k}\):} store values at stock points and \term{interpolate} between them.
\item \textbf{Compute the expectation:} approximate the income integral using \term{numerical quadrature} or sampled incomes.
\item \textbf{Choose saving:} maximize over the feasible interval using numerical optimization or an action grid.
\end{enumerate}
\par\medskip
These approximations add errors beyond the VFI stopping error. Refine the stock grid, integration, and maximization to check that values and decisions are stable.
\par\smallskip
The economic model's states and actions remain continuous; the grid is a computational representation.
\end{frame}
