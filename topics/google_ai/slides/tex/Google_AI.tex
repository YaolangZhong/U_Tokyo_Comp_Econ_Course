\begin{frame}{Google AI: Free Access for Students}
\begin{itemize}
    \item Google offers free/trial access to several AI products
    \item Options:
    \begin{itemize}
        \item \textbf{Google AI Studio}: Free tier with generous limits
        \item \textbf{Google Cloud}: 1-year free trial (\$300 credit) for students
        \item \textbf{Gemini Advanced}: 1-month free trial
    \end{itemize}
    \item Useful for research, coding, and learning
\end{itemize}
\end{frame}

\begin{frame}{How to Register}
\begin{enumerate}
    \item Go to \url{https://aistudio.google.com}
    \item Sign in with your Google account (use university email for student benefits)
    \item For Google Cloud credits:
    \begin{itemize}
        \item Visit \url{https://cloud.google.com/free}
        \item Apply for student credits via \url{https://edu.google.com}
    \end{itemize}
    \item For Gemini Advanced trial:
    \begin{itemize}
        \item Visit \url{https://gemini.google.com}
        \item Click "Try Gemini Advanced" (1-month free)
    \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Product 1: Gemini}
\begin{itemize}
    \item Google's flagship AI assistant (like ChatGPT)
    \item Access via \url{https://gemini.google.com}
    \item Features:
    \begin{itemize}
        \item Text generation and conversation
        \item Code generation and debugging
        \item Image understanding
        \item Integration with Google Workspace
    \end{itemize}
    \item Gemini Advanced: More powerful, 1-month free trial
\end{itemize}
\end{frame}

\begin{frame}{Product 2: Google AI Studio}
\begin{itemize}
    \item Developer platform for Gemini API
    \item Access via \url{https://aistudio.google.com}
    \item Features:
    \begin{itemize}
        \item Free API access with rate limits
        \item Prompt engineering playground
        \item Generate API keys for your applications
        \item Supports text, code, and multimodal inputs
    \end{itemize}
    \item Great for building AI-powered applications
\end{itemize}
\end{frame}

\begin{frame}{Product 3: NotebookLM}
\begin{itemize}
    \item AI-powered research assistant
    \item Access via \url{https://notebooklm.google.com}
    \item Features:
    \begin{itemize}
        \item Upload documents (PDFs, papers, notes)
        \item AI summarizes and answers questions about your sources
        \item Generate study guides and outlines
        \item Audio overview feature (podcast-style summaries)
    \end{itemize}
    \item Perfect for literature review and studying
\end{itemize}
\end{frame}

\begin{frame}{Product 4: Google Colab + AI}
\begin{itemize}
    \item Jupyter notebooks in the cloud
    \item Access via \url{https://colab.research.google.com}
    \item AI Features:
    \begin{itemize}
        \item Code completion with Gemini
        \item Natural language to code
        \item Error explanation and debugging
        \item Free GPU/TPU access (with limits)
    \end{itemize}
    \item Already integrated into our course workflow
\end{itemize}
\end{frame}
